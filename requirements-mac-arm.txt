#
#   1. 创建虚拟环境

#      pip install -r requirements-mac-arm.txt

#   python -c "import torch; print('mps:', torch.backends.mps.is_available())"
#   python -c "import onnxruntime as ort; print(ort.get_available_providers())"

#   mps: True

#
# 说明:
#   - PyTorch 标准版原生支持 MPS 加速，无需额外配置
